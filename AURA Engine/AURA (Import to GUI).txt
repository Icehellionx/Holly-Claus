    function _extractBracketBlock(src, tagName) {
        if (!src) return "";
        var re = new RegExp("\\[" + tagName + "\\]([\\s\\S]*?)\\[\\/" + tagName + "\\]", "i");

        return m ? String(m[1]).trim() : "";

    function _parseScoreBlock(raw) {
        var out = {
            raw: raw || "",
            labels: Object.create(null)
        };
        if (!raw) return out;

        var re = /([a-z0-9_]+)\s*:\s*([0-9]+)%/gi;
        var m;
        while ((m = re.exec(raw)) !== null) {
            var label = String(m[1]).toLowerCase();
            var pct = parseInt(m[2], 10);
            if (!isFinite(pct)) continue;
            out.labels[label] = Math.max(0, Math.min(1, pct / 100));
        }
        return out;
    }

    function _hasWord(haystack, needle) {
        if (!haystack || !needle) return false;
        haystack = haystack.toLowerCase();
        needle = needle.toLowerCase();
        return haystack.indexOf(needle) !== -1;
    }

    function _anyWord(haystack, arr) {
        if (!haystack || !arr || !arr.length) return false;
        haystack = haystack.toLowerCase();
        for (var i = 0; i < arr.length; i++) {
            var w = String(arr[i] || "").toLowerCase();
            if (!w) continue;
            if (haystack.indexOf(w) !== -1) return true;
        }
        return false;
    }

    function _allWords(haystack, arr) {
        if (!haystack || !arr || !arr.length) return false;
        haystack = haystack.toLowerCase();
        for (var i = 0; i < arr.length; i++) {
            var w = String(arr[i] || "").toLowerCase();
            if (!w) continue;
            if (haystack.indexOf(w) === -1) return false;
        }
        return true;
    }

        var start = Math.max(0, CHAT_WINDOW.length - n);
        var acc = "";
        roleMode = roleMode || "any";

        for (var i = start; i < CHAT_WINDOW.length; i++) {
            var msg = CHAT_WINDOW[i];
            if (!msg || !msg.content) continue;

            if (roleMode === "user" && msg.role !== "user") continue;
            if (roleMode === "bot" &&
                !(msg.role === "assistant" || (activeName && msg.name === activeName))) continue;

            acc += " " + msg.content;
        }
        return acc.trim();
    }

    function _stripEngineBlocks(str) {
        if (!str) return "";
        return String(str).replace(/\[(ORBIT|PULSE|SCENE|PACE)\][\s\S]*?\[\/\1\]/gi, "").trim();
    }

    function _cleanupScenarioEngineBlocks() {
        try {
            if (!context || !context.character) return;
            var s = context.character.scenario || "";
            context.character.scenario = _stripEngineBlocks(s);
        } catch (e) {
            if (DEBUG) console.log("[LORE] Error stripping engine blocks:", e);
        }
    }

    // Weighted sampling without replacement — used for inclusion groups
    function _weightedSampleWithoutReplacement(items, weights, k) {
        var selected = [];
        if (!items || !weights || !items.length || !weights.length) return selected;

        var localItems = items.slice();
        var localWeights = weights.slice();
        var n = Math.min(k, localItems.length);

        for (var s = 0; s < n; s++) {
            var total = 0;
            for (var i = 0; i < localWeights.length; i++) {
                var w = localWeights[i];
                if (w > 0) total += w;
            }
            if (total <= 0) break;

            var r = Math.random() * total;
            var cum = 0;
            var idx = -1;
            for (var j = 0; j < localWeights.length; j++) {
                cum += localWeights[j];
                if (r <= cum) {
                    idx = j;
                    break;
                }
            }
            if (idx < 0) idx = localWeights.length - 1;

            selected.push(localItems[idx]);
            localItems.splice(idx, 1);
            localWeights.splice(idx, 1);
        }
        return selected;
    }

    // ---------------------------------------------------------------------------
    // [SECTION] TAG SYSTEM / TRIGGER SET
    // ---------------------------------------------------------------------------

    function _buildTriggerSet() {
        var trig = Object.create(null);

        try {
            if (context && context.character) {
                if (context.character.tags && Array.isArray(context.character.tags)) {
                    for (var i = 0; i < context.character.tags.length; i++) {
                        var rawTag = context.character.tags[i];
                        if (!rawTag) continue;
                        trig[String(rawTag)] = true;
                    }
                }
                if (context.character.personalityTags && Array.isArray(context.character.personalityTags)) {
                    for (var j = 0; j < context.character.personalityTags.length; j++) {
                        var pt = context.character.personalityTags[j];
                        if (!pt) continue;
                        trig[String(pt)] = true;
                    }
                }
            }
        } catch (e) {
            if (DEBUG) console.log("[LORE] Error reading character tags:", e);
        }

        var allText = _getAllVisibleText();
        if (allText) {
            for (var k = 0; k < KEYWORD_TAG_MAP.length; k++) {
                var cfg = KEYWORD_TAG_MAP[k];
                if (!cfg || !cfg.words || !cfg.tag) continue;
                if (_anyWord(allText, cfg.words)) {
                    trig[cfg.tag] = true;
                }
            }
        }

        trig["messages_" + messageCount] = true;
        if (messageCount <= 3) trig["early_session"] = true;
        if (messageCount >= 10) trig["mid_session"] = true;
        if (messageCount >= 20) trig["long_session"] = true;

        var turnCount = messageCount;
        try {
            if (context && context.chat && typeof context.chat.message_count === "number") {
                turnCount = context.chat.message_count;
            }
        } catch (e) { }

        if (turnCount % 2 === 0) {
            trig["turn.user"] = true;
        } else {
            trig["turn.bot"] = true;
        }

        if (CHAT_WINDOW.length > 0) {
            var last = CHAT_WINDOW[CHAT_WINDOW.length - 1];
            if (last.role === "user") {
                trig["last_speaker.user"] = true;
            } else {
                trig["last_speaker.bot"] = true;
            }
        }

        if (typeof LORE_BOOL_TAGS === "object" && LORE_BOOL_TAGS) {
            for (var key in LORE_BOOL_TAGS) {
                if (!Object.prototype.hasOwnProperty.call(LORE_BOOL_TAGS, key)) continue;
                if (LORE_BOOL_TAGS[key]) {
                    trig[key] = true;
                }
            }
        }

        return trig;
    }

    // ---------------------------------------------------------------------------
    // [SECTION] HISTORY STORAGE
    // ---------------------------------------------------------------------------

    function _getLoreHistory() {
        try {
            if (!context || !context.character) return { fired: {}, cooldowns: {}, groups: {} };
            var raw = context.character.loreHistoryJSON || null;
            var parsed = _safeParseJSON(raw, null);
            if (!parsed || typeof parsed !== "object") {
                return { fired: {}, cooldowns: {}, groups: {} };
            }
            if (!parsed.fired || typeof parsed.fired !== "object") parsed.fired = {};
            if (!parsed.cooldowns || typeof parsed.cooldowns !== "object") parsed.cooldowns = {};
            if (!parsed.groups || typeof parsed.groups !== "object") parsed.groups = {};
            return parsed;
        } catch (e) {
            if (DEBUG) console.log("[LORE] Error reading lore history:", e);
            return { fired: {}, cooldowns: {}, groups: {} };
        }
    }

    function _saveLoreHistory(hist) {
        try {
            if (!context || !context.character) return;
            context.character.loreHistoryJSON = JSON.stringify(hist);
        } catch (e) {
            if (DEBUG) console.log("[LORE] Error saving lore history:", e);
        }
    }

    function _hasFired(hist, id) {
        if (!hist || !hist.fired) return false;
        return !!hist.fired[id];
    }

    function _getGroupState(hist, groupId) {
        if (!hist.groups) hist.groups = {};
        if (!hist.groups[groupId]) {
            hist.groups[groupId] = { count: 0, cooldown: 0 };
        }
        return hist.groups[groupId];
    }

    function _groupOnCooldown(hist, groupId) {
        if (!hist || !hist.groups || !hist.groups[groupId]) return false;
        var g = hist.groups[groupId];
        return !!(g && typeof g.cooldown === "number" && g.cooldown > 0);
    }

    function _groupAtMaxUses(hist, groupId, groupMaxUses) {
        if (!groupId || typeof groupMaxUses !== "number") return false;
        if (!hist || !hist.groups || !hist.groups[groupId]) return false;
        var g = hist.groups[groupId];
        return (g.count || 0) >= groupMaxUses;
    }

    function _tickGroupCooldowns(hist) {
        if (!hist || !hist.groups) return;
        var keys = Object.keys(hist.groups);
        for (var i = 0; i < keys.length; i++) {
            var id = keys[i];
            var g = hist.groups[id];
            if (!g || typeof g.cooldown !== "number") continue;
            g.cooldown = g.cooldown - 1;
            if (g.cooldown < 0) g.cooldown = 0;
        }
    }

    function _fire(hist, id, coolDown, maxUses, entry) {
        if (!hist) return;
        if (!hist.fired) hist.fired = {};
        if (!hist.cooldowns) hist.cooldowns = {};
        if (!hist.groups) hist.groups = {};

        var entryState = hist.fired[id] || { count: 0 };
        entryState.count = (entryState.count || 0) + 1;
        if (typeof maxUses === "number") {
            entryState.maxUses = maxUses;
        }
        hist.fired[id] = entryState;

        if (coolDown && coolDown > 0) {
            hist.cooldowns[id] = coolDown;
        }

        // Legacy groupId-based tracking (separate from inclusion groups)
        if (entry && entry.groupId) {
            var g = _getGroupState(hist, entry.groupId);
            g.count = (g.count || 0) + 1;
            if (entry.groupCooldown && entry.groupCooldown > 0) {
                g.cooldown = entry.groupCooldown;
            }
        }
    }

    function _tickCooldowns(hist) {
        if (!hist || !hist.cooldowns) return;
        var keys = Object.keys(hist.cooldowns);
        for (var i = 0; i < keys.length; i++) {
            var id = keys[i];
            var v = hist.cooldowns[id];
            if (typeof v !== "number") continue;
            v = v - 1;
            if (v <= 0) {
                delete hist.cooldowns[id];
            } else {
                hist.cooldowns[id] = v;
            }
        }
    }

    function _isOnCooldown(hist, id) {
        if (!hist || !hist.cooldowns) return false;
        return hist.cooldowns[id] > 0;
    }

    function _hasHitMaxUses(hist, id, maxUses) {
        if (!hist || !hist.fired || typeof maxUses !== "number") return false;
        var entryState = hist.fired[id];
        if (!entryState || typeof entryState.count !== "number") return false;
        return entryState.count >= maxUses;
    }

    // ---------------------------------------------------------------------------
    // [SECTION] ENTRY EVALUATION
    // ---------------------------------------------------------------------------

    function _conditionsMet(entry, trig, hist) {
        if (!entry) return false;

        var minMessages = (typeof entry.minMessages === "number") ? entry.minMessages : 0;
        var maxMessages = (typeof entry.maxMessages === "number") ? entry.maxMessages : 99999;

        if (messageCount < minMessages) {
            return false;
        }
        if (messageCount > maxMessages) {
            return false;
        }

        if (_isOnCooldown(hist, entry.id)) return false;
        if (_hasHitMaxUses(hist, entry.id, entry.maxUses)) return false;

        // Legacy group gating (not the new inclusion-group system)
        if (entry.groupId) {
            if (_groupOnCooldown(hist, entry.groupId)) return false;
            if (_groupAtMaxUses(hist, entry.groupId, entry.groupMaxUses)) return false;

            if (typeof entry.groupFireChance === "number") {
                if (Math.random() > entry.groupFireChance) return false;
            }
        }

        function _hasAllTags(tagArr) {
            for (var i = 0; i < tagArr.length; i++) {
                if (!trig[tagArr[i]]) return false;
            }
            return true;
        }
        function _hasAnyTags(tagArr) {
            for (var i = 0; i < tagArr.length; i++) {
                if (trig[tagArr[i]]) return true;
            }
            return false;
        }

        if (entry.andAllTags && entry.andAllTags.length) {
            if (!_hasAllTags(entry.andAllTags)) return false;
        }
        if (entry.andAnyTags && entry.andAnyTags.length) {
            if (!_hasAnyTags(entry.andAnyTags)) return false;
        }
        if (entry.andNotTags && entry.andNotTags.length) {
            for (var i = 0; i < entry.andNotTags.length; i++) {
                if (trig[entry.andNotTags[i]]) return false;
            }
        }

        var windowSize = entry.keywordWindow;
        var roleMode = entry.keywordRoles || "any";
        var textScope = _getScopedText(windowSize, roleMode) || _getAllVisibleText();

        if (entry.keywordsAny && entry.keywordsAny.length) {
            if (!_anyWord(textScope, entry.keywordsAny)) return false;
        }
        if (entry.keywordsAll && entry.keywordsAll.length) {
            if (!_allWords(textScope, entry.keywordsAll)) return false;
        }

        return true;
    }

    function _computeScore(entry) {
        if (!entry) return -Infinity;

        var base = (typeof entry.priority === "number") ? entry.priority : 5;
        if (base < 1) base = 1;
        if (base > 10) base = 10;

        var randW = (typeof entry.randomWeight === "number") ? entry.randomWeight : 1.0;
        var jitter = Math.random() * randW;

        var gW = (typeof entry.groupWeight === "number") ? entry.groupWeight : 1.0;

        return (base + jitter) * gW;
    }

    // ---------------------------------------------------------------------------
    // [SECTION] MAIN LORE SELECTION (WITH INCLUSION GROUPS)
    // ---------------------------------------------------------------------------

    if (DEBUG) console.log("[LORE] Begin LORE evaluation.");

    var trigSet = _buildTriggerSet();

    if (trigSet[DISABLE_TAG]) {
        if (DEBUG) console.log("[LORE] Disabled via tag:", DISABLE_TAG);
        _cleanupScenarioEngineBlocks();
        return;
    }

    if (messageCount < MIN_CHAT_MESSAGES) {
        if (DEBUG) console.log("[LORE] Not enough messages:", messageCount);
        _cleanupScenarioEngineBlocks();
        return;
    }

    var history = _getLoreHistory();
    _tickCooldowns(history);
    _tickGroupCooldowns(history);

    var eligibleEntries = [];
    for (var ei = 0; ei < LORE_ENTRIES.length; ei++) {
        var entry = LORE_ENTRIES[ei];
        if (!entry || !entry.id) continue;
        if (!_conditionsMet(entry, trigSet, history)) continue;
        eligibleEntries.push(entry);
    }

    if (!eligibleEntries.length) {
        if (DEBUG) console.log("[LORE] No matching entries.");
        _saveLoreHistory(history);
        _cleanupScenarioEngineBlocks();
        return;
    }

    // Partition into:
    //   - ungrouped: no inclusion group
    //   - grouped:   entries keyed by inclusion-group id
    //
    // Inclusion-group ID is taken from:
    //   1) incGroupId (if present)
    //   2) otherwise groupId (GUI field)
    var ungrouped = [];
    var inclusionGroups = Object.create(null);

    for (var iE = 0; iE < eligibleEntries.length; iE++) {
        var e2 = eligibleEntries[iE];
        var gid = e2.incGroupId || e2.groupId || null;

        if (gid) {
            if (!inclusionGroups[gid]) inclusionGroups[gid] = [];
            inclusionGroups[gid].push(e2);
        } else {
            ungrouped.push(e2);
        }
    }

    var winners = [];

    // 1) Pick single best ungrouped entry (preserves old winner-takes-all behavior)
    if (ungrouped.length) {
        var bestUngrouped = null;
        var bestScore = -Infinity;

        for (var u = 0; u < ungrouped.length; u++) {
            var ue = ungrouped[u];
            var sScore = _computeScore(ue);
            if (sScore > bestScore) {
                bestScore = sScore;
                bestUngrouped = ue;
            }
        }

        if (bestUngrouped) {
            winners.push(bestUngrouped);
        }
    }

    // 2) For each inclusion group, roll group chance and then sample up to incGroupMax entries
    for (var gId in inclusionGroups) {
        if (!Object.prototype.hasOwnProperty.call(inclusionGroups, gId)) continue;
        var groupEntries = inclusionGroups[gId];
        if (!groupEntries || !groupEntries.length) continue;

        // Group-level roll probability:
        // Preference order:
        //   1) groupChance (GUI, 0–100)
        //   2) incGroupRoll (0–1 or 0–100)
        //   3) legacy groupFireChance (0–1 or 0–100)
        var rollProb = null;

        // Helper: normalize a number to 0–1, treating >1 as percent.
        function _normalizeProb(v) {
            if (typeof v !== "number" || !isFinite(v)) return null;
            var p = v;
            if (p > 1) p = p / 100; // treat >1 as percent
            if (p < 0) p = 0;
            if (p > 1) p = 1;
            return p;
        }

        // 1) groupChance (0–100)
        for (var gi = 0; gi < groupEntries.length && rollProb == null; gi++) {
            var ge = groupEntries[gi];
            if (typeof ge.groupChance === "number") {
                rollProb = _normalizeProb(ge.groupChance);
            }
        }

        // 2) incGroupRoll (0–1 or percent)
        for (var gj = 0; gj < groupEntries.length && rollProb == null; gj++) {
            var ge2 = groupEntries[gj];
            if (typeof ge2.incGroupRoll === "number") {
                rollProb = _normalizeProb(ge2.incGroupRoll);
            }
        }

        // 3) legacy groupFireChance (0–1 or percent)
        for (var gk = 0; gk < groupEntries.length && rollProb == null; gk++) {
            var ge3 = groupEntries[gk];
            if (typeof ge3.groupFireChance === "number") {
                rollProb = _normalizeProb(ge3.groupFireChance);
            }
        }

        if (rollProb == null) rollProb = 1.0;

        if (Math.random() > rollProb) {
            if (DEBUG) console.log("[LORE] Inclusion group", gId, "failed roll", rollProb);
            continue;
        }

        // Max winners from group:
        //   1) incGroupMax (if present)
        //   2) else groupMaxUses (GUI field)
        //   3) else 1
        var maxFromGroup = null;
        for (var gm = 0; gm < groupEntries.length; gm++) {
            var geMax = groupEntries[gm];
            if (typeof geMax.incGroupMax === "number") {
                maxFromGroup = geMax.incGroupMax;
                break;
            }
            if (maxFromGroup == null && typeof geMax.groupMaxUses === "number") {
                maxFromGroup = geMax.groupMaxUses;
                // don't break here; incGroupMax takes precedence if we see it later
            }
        }
        if (maxFromGroup == null) maxFromGroup = 1;
        if (maxFromGroup < 1) maxFromGroup = 1;

        // Build sampling weights for this group:
        //   weight = priority * randomWeight * (incWeight OR groupWeight)
        var items = [];
        var weights = [];
        for (var gx = 0; gx < groupEntries.length; gx++) {
            var ge4 = groupEntries[gx];

            var p = (typeof ge4.priority === "number") ? ge4.priority : 5;
            if (p < 1) p = 1;
            if (p > 10) p = 10;

            var baseRandW = (typeof ge4.randomWeight === "number") ? ge4.randomWeight : 1.0;
            var baseWeight = p * baseRandW;

            var relW = 1.0;
            if (typeof ge4.incWeight === "number" && ge4.incWeight > 0) {
                relW = ge4.incWeight;
            } else if (typeof ge4.groupWeight === "number" && ge4.groupWeight > 0) {
                relW = ge4.groupWeight;
            }

            var totalWeight = baseWeight * relW;
            if (totalWeight <= 0) continue;

            items.push(ge4);
            weights.push(totalWeight);
        }


        if (!items.length) continue;

        var groupWinners = _weightedSampleWithoutReplacement(items, weights, maxFromGroup);
        for (var gw = 0; gw < groupWinners.length; gw++) {
            winners.push(groupWinners[gw]);
        }
    }

    // Enforce global cap
    if (winners.length > MAX_WINNERS_PER_TURN) {
        // Keep the first N winners; since sampling already has randomness, this is fine.
        winners = winners.slice(0, MAX_WINNERS_PER_TURN);
    }

    if (!winners.length) {
        if (DEBUG) console.log("[LORE] No winners after inclusion-group processing.");
        _saveLoreHistory(history);
        _cleanupScenarioEngineBlocks();
        return;
    }

    if (DEBUG) {
        var ids = [];
        for (var wdi = 0; wdi < winners.length; wdi++) ids.push(winners[wdi].id);
        console.log("[LORE] Winners:", ids.join(", "));
    }

    // ---------------------------------------------------------------------------
    // [SECTION] APPLY WINNER ENTRIES
    // ---------------------------------------------------------------------------

    function _applyNameVars(str) {
        if (!str) return "";
        var userName = "{{user}}";
        var charName = activeName || "{{char}}";

        str = String(str);
        str = str.replace(/\{\{char\}\}/g, charName);
        str = str.replace(/\{\{user\}\}/g, userName);
        return str;
    }

    try {
        if (!context || !context.character) {
            if (DEBUG) console.log("[LORE] No context.character to write into.");
        } else {
            for (var wi = 0; wi < winners.length; wi++) {
                var winner = winners[wi];
                if (!winner || !winner.text) continue;

                var outText = _applyNameVars(winner.text);
                if (!outText) continue;

                if (outText.length > MAX_APPEND_CHARS) {
                    outText = outText.slice(0, MAX_APPEND_CHARS);
                }

                var tgt = (winner.target === "personality") ? "personality" : "scenario";

                if (tgt === "personality") {
                    var existingP = context.character.personality || "";
                    context.character.personality = existingP
                        ? existingP + "\n\n" + outText
                        : outText;
                } else {
                    var existingS = context.character.scenario || "";
                    existingS = _stripEngineBlocks(existingS);
                    context.character.scenario = existingS
                        ? existingS + "\n\n" + outText
                        : outText;
                }

                _fire(history, winner.id, winner.coolDown, winner.maxUses, winner);

                if (winner.id === "session_intro" || winner.id === "flag_session_intro_done") {
                    history.fired["session_intro_done"] =
                        history.fired["session_intro_done"] || { count: 0 };
                }
            }
        }
    } catch (e) {
        if (DEBUG) console.log("[LORE] Error applying LORE winners:", e);
    }

    _saveLoreHistory(history);
    _cleanupScenarioEngineBlocks();

    if (DEBUG) console.log("[LORE] Done.");

})();
